cd /home/jovis/Documents/WORK/BleBelTranslateBook
source .venv_splitter/bin/activate

# Install the package itself (the extras are convert/dictionary/chunk/epub/dev):
#   pip install -e ".[dev]"              # development: tests, lint, types
#   pip install -e ".[convert]"          # EPUB/PDF input, or "book-splitter[convert]"
# The installed entry points are `book-splitter` and `book-splitter-pipeline`;
# `python -m book_splitter` remains the module fallback.

# Whole pipeline: `python -m book_splitter pipeline ...` is the same parser and
# driver as `python scripts/run_pipeline.py ...` (the script is a wrapper).
# Shared flags: --json (one object, no narration), --quiet (no narration),
# --output-root (default data/output), --text-in (chapter text override).
# --dry-run validates and writes nothing; it never calls a paid model.
#   .venv_splitter/bin/python -m book_splitter pipeline BOOK.md --json
#   .venv_splitter/bin/python -m book_splitter pipeline BOOK.md --dry-run --quiet

# which model each stage uses: book_splitter.toml ([llm] + [llm.<stage>]).
# Override one stage for one run without editing it:
#   --llm split.model=deepseek-flash --llm translate.provider=lmstudio
# Or point at another per-stage file:
#   --llm-config /tmp/cheap.toml        (or BOOK_SPLITTER_CONFIG=...)
# Keys live in .env (DEEPSEEK_API_KEY, OPENAI_API_KEY, ...), never in the toml.

# stage 0: EPUB/text-based PDF/TXT -> Markdown in data/output/<book>/00-markdown/
# (TXT is copied; a PDF with no embedded text layer is rejected instead of OCR'd)
# Pictures are extracted to 00-markdown/images/ by default; add --no-images to skip.
.venv_splitter/bin/python -m book_splitter convert "My Book.epub"
.venv_splitter/bin/python -m book_splitter convert "My Book.epub" --no-images
.venv_splitter/bin/python -m book_splitter convert "My Book.pdf" --dry-run
.venv_splitter/bin/python scripts/run_pipeline.py "My Book.epub" --stop-after convert

# the whole pipeline over the first N chapters (split itself is always the full book);
# --skip-if-present makes a later, larger N reuse the finished chapters and chunks.
# Stage 2 (preprocess) is on by default and its cleaned 02-preprocessed/ folder is
# what the dictionary, glossary and chunker read: add --no-preprocess to skip the
# cleaning (and read 01-chapters/), or --text-in <folder> to choose explicitly.
.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --skip-if-present \
  --first-chapters 5 \
  --cpu

# same run, and translate + assemble the first five chapters
.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --skip-if-present \
  --first-chapters 5 \
  --with-translate \
  --cpu

# translated chapters + heading postprocessing + one full book
.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --skip-if-present \
  --first-chapters 5 \
  --with-translate \
  --with-postprocess \
  --cpu

# the same translation, but stage 7 uses the LOCAL LM Studio model instead of
# DeepSeek. Only the body-chapter translation changes: front/back matter (stage
# 6b) and every other stage keep their model from book_splitter.toml.
# The same override without editing the file:
#   --llm translate.provider=lmstudio --llm translate.model=<id>
# Omit --translate-model to auto-detect the loaded model (an "en2sr" id wins).
.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --skip-if-present \
  --first-chapters 5 \
  --with-translate \
  --translate-provider lmstudio \
  --translate-model gemma-4-31b-en2sr-v3@q4_k_m \
  --cpu

# stage 7 only, with the local model, over existing chunks
.venv_splitter/bin/python -m book_splitter translate \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --translator lmstudio \
  --model gemma-4-31b-en2sr-v3@q4_k_m

# stage 7 only, explicitly back on DeepSeek (the default)
.venv_splitter/bin/python -m book_splitter translate \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --translator deepseek

# stage 6b only: localize front/back matter (no dictionary; web-searches the
# published title; credits BleBel; omits original-edition imprint material).
# Missing 00-front-matter.md / 99-back-matter.md are created from the filename
# ("Title - Author") and localized too.
.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --start-at matter \
  --cpu

# stage 8 only, over an existing translation
.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --start-at concatenate \
  --first-chapters 5 \
  --skip-if-present \
  --cpu

# stage 9 only, then stage 10 only
.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --start-at postprocess \
  --first-chapters 5 \
  --skip-if-present \
  --cpu

.venv_splitter/bin/python scripts/run_pipeline.py \
  "Dont You Forget About Me - Alexandra Potter.md" \
  --start-at bind \
  --cpu

# standalone commands
.venv_splitter/bin/python -m book_splitter matter \
  "Dont You Forget About Me - Alexandra Potter.md" --resume
.venv_splitter/bin/python -m book_splitter postprocess \
  "Dont You Forget About Me - Alexandra Potter.md" --resume

# the final opening audit of stages 2 and 9 (no LLM, no key). It walks the first
# lines of every chapter, removes a stray copy of the chapter number under the
# title, a duplicated title line or a leftover page number, and reports the rest.
# Default folder: the book's most advanced stage (09 -> 08 -> 02 -> 01).
.venv_splitter/bin/python -m book_splitter openings \
  "Dont You Forget About Me - Alexandra Potter.md" --dry-run
.venv_splitter/bin/python -m book_splitter openings \
  "Dont You Forget About Me - Alexandra Potter.md"
# a stage-2 folder: a removed line leaves an empty line, so line numbers stay valid
.venv_splitter/bin/python -m book_splitter openings \
  --chapters-in data/output/BOOK/02-preprocessed --keep-blank-slot

.venv_splitter/bin/python -m book_splitter bind \
  "Dont You Forget About Me - Alexandra Potter.md"
